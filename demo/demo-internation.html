Tablechart

for Reveal.js: Internation

Translate a chart

Internation translates the names in a chart like any other text on the slide. Numbers are formatted for the chosen language.

Markup

Add data-chart-keys to the figure. Tablechart then adds translation keys to the rows before Reveal starts, so Internation can find them.

<figure data-chart="column"
        data-chart-keys="intl">
  <figcaption>
    <b data-i18n="caption">Seeds per fruit</b>
  </figcaption>
  <table>
    <tbody>
      <!-- intl-row1, intl-row2, intl-row3
           are written on these by the plugin -->
      <tr><th>Apples</th><td>8.2</td></tr>
    </tbody>
  </table>
</figure>
Reveal.initialize({
  plugins: [ Tablechart, Internation ],
  internation: {
    languages: {
      nl: { name: "Nederlands",
            dictionary: "tablechart-nl.json" }
    }
  }
});
